#!/usr/bin/env python3
"""Disposable E6 reporting probe (#43): frozen May 2015 and NIST SRD 69 replays through the
installed public interface, plus the equivalent-work timing protocol. Writes one native-format
comparison log (EPCSAFT_COMPARE_JSON lines) and one timing JSON. No Engine source is imported."""
import csv, hashlib, json, math, platform, statistics, subprocess, sys, time
from pathlib import Path

import epcsaft
from epcsaft import Mixture, Parameters, Pure
from epcsaft.equilibrium import Amounts, Continuation, Free, Phase, Pinned, Problem, continue_equilibrium, solve_equilibrium

ROOT = Path(sys.argv[1]).resolve()
LOG = open(sys.argv[2], "w")
TIMING = Path(sys.argv[3])
BUILD = Path(sys.argv[4]) if len(sys.argv) > 4 else None
PARAMS = ROOT / "data/packets/gross-2001-methane-ethane/2/parameters/parameters.json"
MAY = ROOT / "analyses/reproduction/2015-may/inputs/source/may-2015-methane-ethane-vle.csv"
NIST = ROOT / "data/packets/nist-srd69-pure-saturation-properties/1/observations/ethane-saturation.csv"
ROW_TOLERANCE = 1.0e-10
M_ETHANE = 0.03007  # kg/mol, packet gross-2001-methane-ethane v2 ethane molar mass


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def finite(value):
    return value is not None and isinstance(value, (int, float)) and math.isfinite(value)


def row(cid, quantity, unit, model, reference, inputs, basis, method, claim, criterion=None, atol=0.0, rtol=0.0,
        verdict=None, group=None, point=None, evidence_state=None, reason=None):
    ok = finite(model) and (reference is None or finite(reference))
    payload = {"schema_version": 1, "comparison_id": cid, "quantity": quantity, "unit": unit,
               "model": model if finite(model) else None}
    if reference is not None:
        payload["reference"] = reference if finite(reference) else None
    payload.update({"inputs": inputs, "basis": basis, "method": method, "claim": claim,
                    "judgment": ("Not evaluated" if evidence_state == "Not evaluated" else "Observation only" if criterion is None
                                 else "Criterion satisfied" if verdict else "Criterion failed"),
                    "evidence_state": evidence_state or ("Complete" if ok else "Unavailable")})
    if reason or not ok:
        payload["unavailable_reason"] = reason or "nonfinite comparison operand"
    if criterion is not None:
        payload.update({"criterion": criterion, "rule": criterion, "atol": atol, "rtol": rtol, "verdict": bool(verdict)})
    if group:
        payload["comparison_group_id"] = group
    if point:
        payload["point_id"] = point
    LOG.write("EPCSAFT_COMPARE_JSON " + json.dumps(payload) + "\n")


def diagnostics(result):
    conv = result.convergence
    return {"success": result.success, "solver_status": result.solver_status, "message": result.message,
            "requested_tolerance_met": result.requested_tolerance_met, "stop": str(conv.stop),
            "classified_residual": conv.residual, "criterion": conv.criterion.value if conv.criterion is not None else None,
            "floor": conv.floor, "unscaled_residual": max((abs(v) for v in result.residuals), default=None),
            "iterations": result.iterations, "evaluations": result.evaluations, "hessian_calls": result.hessian_calls,
            "phase_count": result.phase_count, "topology_event": str(result.topology_event),
            "amounts": list(result.amounts), "molar_densities": list(result.molar_densities),
            "mole_fractions": list(result.mole_fractions), "pressure": list(result.pressure), "temperature": list(result.temperature)}


PARAMETER_INPUTS = {"parameter_file": str(PARAMS.relative_to(ROOT)), "parameter_sha256": sha(PARAMS),
                    "components": ["methane", "ethane"], "m": [1.0, 1.6069], "sigma_angstrom": [3.7039, 3.5206],
                    "epsilon_over_k_K": [150.03, 191.42], "molar_mass_kg_per_mol": [0.016043, 0.03007], "k_ij": 0.0, "l_ij": 0.0,
                    "public_call": "epcsaft.Parameters.from_json(path, components=order); Mixture/Pure; equilibrium.Problem; solve_equilibrium(model, problem)"}


def may_replay(model):
    inputs_base = dict(PARAMETER_INPUTS, source_csv=str(MAY.relative_to(ROOT)), source_sha256=sha(MAY),
                       metadata_sha256=sha(MAY.with_suffix(".yaml")), row_tolerance=ROW_TOLERANCE,
                       construction="fresh fixed-TP Problem, L Liquid free, V Vapor free, full support, feed z=0.5*(x+y) (feed construction only), guesses = source x/y, no start")
    counts = {"Complete": 0, "Failed": 0, "Non-admission": 0, "reference_miss": 0}
    with open(MAY) as handle:
        rows = list(csv.DictReader(handle))
    assert len(rows) == 17
    for r in rows:
        T, P = float(r["T_K"]), float(r["P_Pa"])
        x = [float(r["x_methane"]), float(r["x_ethane"])]
        y = [float(r["y_methane"]), float(r["y_ethane"])]
        z = [0.5 * (x[i] + y[i]) for i in range(2)]
        problem = Problem(phases=[Phase("L", kind="liquid", composition_guess=x), Phase("V", kind="vapor", composition_guess=y)],
                          T=T, P=P, feed=Amounts({"methane": z[0], "ethane": z[1]}), row_tolerance=ROW_TOLERANCE)
        result = solve_equilibrium(model, problem)
        diag = diagnostics(result)
        admitted = result.success and bool(result.requested_tolerance_met) and result.phase_count == 2 \
            and all(a > 0.0 for a in result.amounts) and result.topology_phase is None and result.topology_pair is None
        # The driver's topology gate (phases_coincide, vanished phase) is Non-admission with its
        # named reason; an unmet numerical criterion or solver failure is Failed.
        topology_rejected = (not result.success) and bool(result.requested_tolerance_met) and "phases_coincide" in result.message
        state = "Complete" if admitted else ("Non-admission" if topology_rejected else "Failed")
        inputs = dict(inputs_base, row_id=r["row_id"], pathway=r["pathway"], T_K=T, P_Pa=P, feed_z=z, source_x=x, source_y=y,
                      uc_x_methane=float(r["uc_x_methane"]), uc_y_methane=float(r["uc_y_methane"]),
                      x_allowance=float(r["x_comparison_allowance"]), y_allowance=float(r["y_comparison_allowance"]),
                      admission=("admitted" if admitted else "rejected"), **diag)
        row(f"ref-may/{r['row_id']}/solve", "classified_residual", "1", diag["classified_residual"], None, inputs,
            "Independent reference", "public-solve", "fresh fixed-TP two-phase solve of the frozen May source row; admission by the current driver",
            group="ref-may", point=r["row_id"], evidence_state=state,
            reason=None if admitted else (f"driver topology gate: {result.message}" if topology_rejected else f"solve not admitted: {result.message}"))
        if not admitted:
            counts[state] += 1

def timed(label, fn, warmups=3, repeats=7, counters=None):
    for _ in range(warmups):
        fn()
    samples, failures, last = [], 0, None
    for _ in range(repeats):
        t0 = time.perf_counter()
        last = fn()
        samples.append(time.perf_counter() - t0)
        if last is not None and hasattr(last, "success") and not last.success:
            failures += 1
        if last is not None and hasattr(last, "returncode") and last.returncode != 0:
            failures += 1
    record = {"row": label, "warmups": warmups, "repeats": repeats, "median_s": statistics.median(samples), "min_s": min(samples),
              "max_s": max(samples), "failures": failures, "counters": counters(last) if counters and last is not None else None}
    row(f"work/{label}/median_seconds", "wall_time", "s", record["median_s"], None, record, "Cost", "timing",
        "equivalent-work wall time on one numerical thread (observation only; no threshold)", group="work", point=label)
    return record
